# Геометрия и управление реальным Jetbot 5 (12.10.2026)
# Объектив на 0.05 м впереди оси ведущих колёс, наклон вниз примерно 15 градусов.
# Камера предполагается по центру, без бокового поворота и крена.
# Высота не измерена: в конфигурации z=0, для плоского управления используются x/y.
# Эта конфигурация не является полной измеренной 3D-калибровкой TF.
# camera_rotation_rpy=[-1.8325957145940461, 0.0, -1.5707963267948966]
# Углы относятся к преобразованию оптических осей в оси робота,
# поэтому первый угол -105 градусов, хотя физический наклон вниз — 15 градусов.
# x_robot = cos(15 deg)*z_camera - sin(15 deg)*y_camera + 0.05
# y_robot = -x_camera
# stop_distance=0.5 м: горизонтальное расстояние от середины оси колёс до цели.
# /target/distance и подпись range: 3D-расстояние от камеры до маркера.
# /target/base_distance: плоское расстояние от середины оси колёс, для регулятора.
# Проверки рулеткой от камеры: 20->~20, 40->40, 60->62, 80->82 см.
# max_linear_speed=0.05 м/с, max_angular_speed=0.3 рад/с; enabled=false.
# Файл: ros2_ws/src/jetbot_target_follower/config/follower_jetbot5.yaml

# В контейнере Jetbot; камера и target_detector должны работать отдельно.
# Перед повторным запуском завершить предыдущий control_jetbot5.launch.py.
cd /home/app/ros2_ws
source install/setup.bash
export ROS_DOMAIN_ID=5 ROS_LOCALHOST_ONLY=0
ros2 launch jetbot_target_follower control_jetbot5.launch.py
# По умолчанию выход — /jetbot_hw1/cmd_vel_preview, к моторам не подключён.
# Это запускает target_follower, twist_mux и command_router.
# Камера, распознавание и моторный драйвер этим launch НЕ запускаются.

# В другом терминале контейнера Jetbot:
ros2 topic echo /follower/state
ros2 topic echo /target/base_distance
ros2 topic echo /jetbot_hw1/cmd_vel_preview
# Пока enabled=false, автономный выход должен быть нулевым (DISABLED).
# Ручной источник имеет отдельный приоритет и не отключается параметром enabled.

# Для последующего подключения моторного драйвера целевой топик:
# /diffbot_base_controller/cmd_vel_unstamped
# Launch принимает аргумент cmd_vel_topic для выбора конечного топика.
# На этапе проверки геометрии оставляем cmd_vel_topic по умолчанию.
# Не запускать одновременно штатный activate_all_drivers (с его twist_mux)
# и второй mux, оба пишущие в один моторный топик.
# Первую проверку направлений колёс проводить с вывешенными колёсами.
# Сохранённый параметр драйвера cmd_vel_timeout=0.5 с ещё нужно проверить в работе.
# Остановка автономного управления (не отменяет отдельные ручные команды):
ros2 param set /target_follower enabled false
